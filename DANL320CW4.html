titanic <- titanic |>

dtrain <- titanic |> filter(rnd >= 0.20) |> select(-rnd)
dtest  <- titanic |> filter(rnd < 0.20)

model2 <- glm(
  survived ~ gender + class + age + fare,
  data = dtrain,
  family = binomial
)

Confusion Matrix

threshold <- 0.02

df_cm <- model2 |>
  augment(newdata = dtest,
          type.predict = "response") |> 
  mutate(
    actual = factor(survived, 
                    levels = c(0, 1), 
                    labels = c("not survived", "survived")),
    pred   = factor(if_else(.fitted > threshold, 1, 0),
                    levels = c(0, 1), 
                    labels = c("not survived", "survived"))
  )

conf_mat <- table(truth = df_cm$actual,
                  prediction = df_cm$pred)
